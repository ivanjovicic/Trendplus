Private Sub cmdIzbrisi_Click()
    lngIDDnevnik = lstDnevnik.Column(0, lstDnevnik.ListIndex + 1)
    lngRedniBroj = lstDnevnik.Column(1, lstDnevnik.ListIndex + 1)
    dateDatum = lstDnevnik.Column(2, lstDnevnik.ListIndex + 1)
    strBrojKalkulacije = lstDnevnik.Column(4, lstDnevnik.ListIndex + 1)
    VratiUnazad lngRedniBroj
    boolIzmena = True
    strTipPromene = lstDnevnik.Column(3, lstDnevnik.ListIndex + 1)
    
    If MsgBox("Obrisati promenu " & strTipPromene & "(" & strBrojKalkulacije & ")" & " od " & dateDatum, vbYesNo, "Brisanje!") = vbYes Then
        
        Select Case strTipPromene
            Case "PRODAJA"
                Dim rstIzbrisiProdaja As New ADODB.Recordset
                rstIzbrisiProdaja.Open "SELECT * FROM tblProdaja WHERE IDDnevnik=" & lngIDDnevnik, cnnVeza, adOpenKeyset, adLockOptimistic
                Do Until rstIzbrisiProdaja.EOF
                    lngIDArtikal = rstIzbrisiProdaja!IDArtikal
                    If PrikaziPromene_UPPP("PONISTI") = -1 Then
                    End If
                    rstIzbrisiProdaja.MoveNext
                Loop
                DoCmd.RunSQL "DELETE * FROM tblProdaja WHERE IDDnevnik=" & lngIDDnevnik
                rstIzbrisiProdaja.Close
            Case "UNOS ROBE"
                Dim rstIzbrisiUnos As New ADODB.Recordset
                rstIzbrisiUnos.Open "SELECT * FROM tblUnosRobe WHERE IDDnevnik=" & lngIDDnevnik, cnnVeza, adOpenKeyset, adLockOptimistic
                Do Until rstIzbrisiUnos.EOF
                    lngIDArtikal = rstIzbrisiUnos!IDArtikal
                    If PrikaziPromene_UPPP("PONISTI") = -1 Then
                    End If
                    rstIzbrisiUnos.MoveNext
                Loop
                DoCmd.RunSQL "DELETE * FROM tblUnosRobe WHERE IDDnevnik=" & lngIDDnevnik
                rstIzbrisiUnos.Close
            Case "PRENOS ROBE"
                Dim rstIzbrisiPrenos As New ADODB.Recordset
                rstIzbrisiPrenos.Open "SELECT * FROM tblPrenosRobe WHERE IDDnevnik=" & lngIDDnevnik, cnnVeza, adOpenKeyset, adLockOptimistic
                Do Until rstIzbrisiPrenos.EOF
                    lngIDArtikal = rstIzbrisiPrenos!IDArtikal
                    If PrikaziPromene_UPPP("PONISTI") = -1 Then
                    End If
                    rstIzbrisiPrenos.MoveNext
                Loop
                DoCmd.RunSQL "DELETE * FROM tblPrenosRobe WHERE IDDnevnik=" & lngIDDnevnik
                DoCmd.RunSQL "DELETE * FROM tblUnosRobe WHERE IDDnevnik=" & lngIDDnevnik
                rstIzbrisiPrenos.Close
            Case "NIVELACIJA"
                Dim rstIzbrisiNivelacija As New ADODB.Recordset
                rstIzbrisiNivelacija.Open "SELECT * FROM tblNivelacije WHERE IDDnevnik=" & lngIDDnevnik, cnnVeza, adOpenKeyset, adLockOptimistic
                Do Until rstIzbrisiNivelacija.EOF
                    lngIDArtikal = rstIzbrisiNivelacija!IDArtikal
                    curStaraCena = rstIzbrisiNivelacija!StaraCena
                    If PrikaziPromene_Nivelacija("PONISTI") = -1 Then
                    End If
                    rstIzbrisiNivelacija.MoveNext
                Loop
                DoCmd.RunSQL "DELETE * FROM tblNivelacije WHERE IDDnevnik=" & lngIDDnevnik
                rstIzbrisiNivelacija.Close
            Case "POVRATNICA"
                Dim rstIzbrisiPovratnica As New ADODB.Recordset
                rstIzbrisiPovratnica.Open "SELECT * FROM tblPovratnice WHERE IDDnevnik=" & lngIDDnevnik, cnnVeza, adOpenKeyset, adLockOptimistic
                Do Until rstIzbrisiPovratnica.EOF
                    lngIDArtikal = rstIzbrisiPovratnica!IDArtikal
                    If PrikaziPromene_UPPP("PONISTI") = -1 Then
                    End If
                    rstIzbrisiPovratnica.MoveNext
                Loop
                DoCmd.RunSQL "DELETE * FROM tblPovratnice WHERE IDDnevnik=" & lngIDDnevnik
                rstIzbrisiPovratnica.Close
        End Select
        
        DoCmd.RunSQL "DELETE * FROM tblDnevnikPromena WHERE IDDnevnik=" & lngIDDnevnik
        lstDnevnik.Requery
        PrikaziNaslov
    
    End If
    
    Sadasnjost lngRedniBroj
    boolIzmena = False
End Sub

Private Sub cmdIzmeni_Click()
    lngIDDnevnik = lstDnevnik.Column(0, lstDnevnik.ListIndex + 1)
    lngRedniBroj = lstDnevnik.Column(1, lstDnevnik.ListIndex + 1)
    dateDatum = lstDnevnik.Column(2, lstDnevnik.ListIndex + 1)
    strBrojKalkulacije = lstDnevnik.Column(4, lstDnevnik.ListIndex + 1)
    strNapomena = lstDnevnik.Column(6, lstDnevnik.ListIndex + 1)
    VratiUnazad lngRedniBroj
    boolIzmena = True
    strTipPromene = lstDnevnik.Column(3, lstDnevnik.ListIndex + 1)
    
    Select Case strTipPromene
        Case "NIVELACIJA"
            DoCmd.OpenForm "frmNivelacija", , , , , acDialog, CStr(lngRedniBroj)
        Case "PRENOS ROBE"
            Dim strPrenosIz As String, strPrenosU As String
            rstDnevnik.Open "SELECT * FROM tblDnevnikPromena ORDER BY RedniBroj", cnnVeza, adOpenKeyset, adLockOptimistic
            rstDnevnik.Find "RedniBroj=" & CStr(lstDnevnik.Column(1, lstDnevnik.ListIndex + 1))
            For intBrojac = 1 To 2
                If rstDnevnik!TipPromene = "PRENOS ROBE" Then
                    strPrenosIz = CStr(rstDnevnik!IDObjekat)
                ElseIf rstDnevnik!TipPromene = "UNOS ROBE" Then
                    strPrenosU = CStr(rstDnevnik!IDObjekat)
                End If
                rstDnevnik.MoveNext
            Next intBrojac
            rstDnevnik.Close
            DoCmd.OpenForm "frmPrenosRobe", , , , , acDialog, CStr(lstDnevnik.Column(0, lstDnevnik.ListIndex + 1)) & ";" & strPrenosIz & ";" & strPrenosU
        Case "UNOS ROBE"
            DoCmd.OpenForm "frmUnosArtikal", , , , , acDialog
        Case "PRODAJA"
            DoCmd.OpenForm "frmProdaja", , , , , acDialog
        Case "POVRATNICA"
            DoCmd.OpenForm "frmProdaja", , , , , acDialog
    End Select
    
    lstDnevnik.Requery
    PrikaziNaslov
    
End Sub

Private Sub lstDnevnik_DblClick(Cancel As Integer)
   
    rstDnevnik.Open "SELECT * FROM tblDnevnikPromena WHERE RedniBroj=" & _
    CStr(lstDnevnik.Column(1, lstDnevnik.ListIndex + 1)) & " AND IDObjekat=" & lngIDObjekat, cnnVeza, adOpenKeyset, adLockOptimistic
            
    If IsShiftKeyDown = False Then
        strBrojKalkulacije = rstDnevnik!BrojKalkulacije
        Dim strNovBrojKalkulacije As String
        strNovBrojKalkulacije = InputBox("Broj kalkulacije:", "Promena broja kalkulacije", strBrojKalkulacije)
        If Not strNovBrojKalkulacije = "" Then
            strBrojKalkulacije = strNovBrojKalkulacije
            rstDnevnik!BrojKalkulacije = strBrojKalkulacije
            rstDnevnik.Update
            lstDnevnik.Requery
        End If
    Else
        If Not IsNull(rstDnevnik!Napomena) Then
            strKomentar = rstDnevnik!Napomena
        Else
            strKomentar = ""
        End If
        Dim strNovKomentar As String
        strNovKomentar = InputBox("Komentar:", "Promena komentara", strKomentar)
        If Not strNovKomentar = "" Then
            strKomentar = strNovKomentar
            rstDnevnik!Napomena = strKomentar
            rstDnevnik.Update
            lstDnevnik.Requery
        End If
    End If
        
    rstDnevnik.Close
    
End Sub

Private Sub PrikaziNaslov()
            
    Dim rstDnevnik As New ADODB.Recordset
    rstDnevnik.Open "SELECT SUM(IznosPromene) AS DnevnikVrednost FROM tblDnevnikPromena WHERE IDObjekat=" & lngIDObjekat, cnnVeza, adOpenKeyset, adLockOptimistic
    On Error Resume Next
    Form.Caption = "Dnevnik promena za objekat " & NazivObjekta & "; Stanje: " & Format(CStr(rstDnevnik!DnevnikVrednost), "currency")
    rstDnevnik.Close

End Sub
