Private Sub cmdFiskalna_Click()
    '1   311171  000000000000    850.00  €   0   0   1.000   0.00    0.000
    'IDArtikal, Artikal, PLU, Naziv, Dobavljac, Kolicina, ProdajnaCena, NabavnaCena, Komentar
    Dim strPutanja As String
    Dim dlgSaveFile As FileDialog
    Set dlgSaveFile = Application.FileDialog(msoFileDialogSaveAs)
    
    With dlgSaveFile
        .Title = "Izvezi u fajl za Galeb fiskalnu kasu"
        .AllowMultiSelect = False
        .InitialFileName = strExportPutanja
        If .Show = True Then
            strPutanja = .SelectedItems(1)
            If InStr(strPutanja, ".") = 0 Then strPutanja = strPutanja & ".txt"
        Else
            Exit Sub
        End If
    End With
    
    
    Open strPutanja For Output As #14
    For intBrojac = 1 To lstPregled.ListCount - 1
        If Not Val(lstPregled.Column(2, intBrojac)) = 0 Then
            If Len(lstPregled.Column(1, intBrojac) & " " & lstPregled.Column(3, intBrojac)) <= 15 Then
                Print #14, lstPregled.Column(2, intBrojac) & Chr(9) & lstPregled.Column(1, intBrojac) & " " & lstPregled.Column(3, intBrojac) & Chr(9) & "000000000000" & Chr(9) & lstPregled.Column(6, intBrojac) & Chr(9) & "4" & Chr(9) & "0" & Chr(9) & "0" & Chr(9) & lstPregled.Column(5, intBrojac) & Chr(9) & "0.00" & Chr(9) & "0.000"
            Else
                Print #14, lstPregled.Column(2, intBrojac) & Chr(9) & Left(lstPregled.Column(1, intBrojac) & " " & lstPregled.Column(3, intBrojac), 15) & Chr(9) & "000000000000" & Chr(9) & lstPregled.Column(6, intBrojac) & Chr(9) & "4" & Chr(9) & "0" & Chr(9) & "0" & Chr(9) & lstPregled.Column(5, intBrojac) & Chr(9) & "0.00" & Chr(9) & "0.000"
            End If
        End If
    Next intBrojac
    Close #14
End Sub

Private Sub Prikazi()

    
    
    'string koji cu kasnije da iskoristim u SQLu za sortiranje, u zavisnosti od izabranog u combo boxu
    Select Case cmbSortiraj.Value
        Case "Artikal"
            strSortiraj = "Artikal,ProdajnaCena"
        Case "Sifra"
            strSortiraj = "PLU,Artikal"
        Case "Tip obuce"
            strSortiraj = "Naziv,Artikal"
        Case "Dobavljac"
            strSortiraj = "Dobavljac,Artikal"
        Case "Kolicina"
            strSortiraj = "Kolicina,Artikal"
        Case "Prodajna cena"
            strSortiraj = "ProdajnaCena,Artikal"
        Case "Nabavna cena"
            strSortiraj = "NabavnaCena,Artikal"
        Case "Sezona"
            strSortiraj = "Sezona,Artikal"
    End Select
    
    'SQL upit: povezane tabele Artikli i Dobavljaci da bi u prikazu dobio imena dobavljaca, a ne ID-jeve
    'takodje sortiram tabelu u zavisnosti od izabrane opcije
    Dim strUpit As String
    If boolIzmena = False And lngRedniBroj = lngMaxRedniBroj Then
        If chkLager.Value = True Then
            strUpit = "SELECT IDArtikal, Artikal, PLU, Naziv, Dobavljac, Kolicina, ProdajnaCena, NabavnaCena, Sezona, Komentar" & _
            " FROM ((tblArtikli INNER JOIN tblDobavljaci ON tblArtikli.IDDobavljac = tblDobavljaci.IDDobavljac)" & _
            " INNER JOIN tblTipObuce ON tblArtikli.IDTipObuce=tblTipObuce.IDTipObuce)" & _
            " INNER JOIN tblSezona ON tblArtikli.IDSezona=tblSezona.IDSezona" & _
            " WHERE IDObjekat=" & lngIDObjekat & " AND Kolicina>0 ORDER BY " & strSortiraj
        Else
            strUpit = "SELECT IDArtikal, Artikal, PLU, Naziv, Dobavljac, Kolicina, ProdajnaCena, NabavnaCena, Sezona, Komentar" & _
            " FROM ((tblArtikli INNER JOIN tblDobavljaci ON tblArtikli.IDDobavljac = tblDobavljaci.IDDobavljac)" & _
            " INNER JOIN tblTipObuce ON tblArtikli.IDTipObuce=tblTipObuce.IDTipObuce)" & _
            " INNER JOIN tblSezona ON tblArtikli.IDSezona=tblSezona.IDSezona" & _
            " WHERE IDObjekat=" & lngIDObjekat & " ORDER BY " & strSortiraj
        End If
    Else
        If chkLager.Value = True Then
            strUpit = "SELECT IDArtikal, Artikal, PLU, Naziv, Dobavljac, Kolicina, ProdajnaCena, NabavnaCena, Sezona, Komentar" & _
            " FROM ((tblArtikliProslost INNER JOIN tblDobavljaci ON tblArtikliProslost.IDDobavljac = tblDobavljaci.IDDobavljac)" & _
            " INNER JOIN tblTipObuce ON tblArtikliProslost.IDTipObuce=tblTipObuce.IDTipObuce)" & _
            " INNER JOIN tblSezona ON tblArtikliProslost.IDSezona=tblSezona.IDSezona" & _
            " WHERE IDObjekat=" & lngIDObjekat & " AND Kolicina>0 ORDER BY " & strSortiraj
        Else
            strUpit = "SELECT IDArtikal, Artikal, PLU, Naziv, Dobavljac, Kolicina, ProdajnaCena, NabavnaCena, Sezona, Komentar" & _
            " FROM ((tblArtikliProslost INNER JOIN tblDobavljaci ON tblArtikliProslost.IDDobavljac = tblDobavljaci.IDDobavljac)" & _
            " INNER JOIN tblTipObuce ON tblArtikliProslost.IDTipObuce=tblTipObuce.IDTipObuce)" & _
            " INNER JOIN tblSezona ON tblArtikliProslost.IDSezona=tblSezona.IDSezona" & _
            " WHERE IDObjekat=" & lngIDObjekat & " ORDER BY " & strSortiraj
        End If
    End If
    
    Form.RecordSource = strUpit
    lstPregled.RowSource = strUpit
    PrikaziNaslov

End Sub
Private Sub cmdIzvestaj_Click()
On Error GoTo Err_cmdIzvestaj_Click

    Dim stDocName As String
    
    If lngRedniBroj = lngMaxRedniBroj Then
        stDocName = "rptPopis"
        DoCmd.OpenReport stDocName, acViewPreview, , "tblArtikli.IDObjekat=" & lngIDObjekat & " AND tblArtikli.Kolicina>0", acDialog, strSortiraj
    Else
        stDocName = "rptPopisProslost"
        DoCmd.OpenReport stDocName, acViewPreview, , "tblArtikliProslost.IDObjekat=" & lngIDObjekat & " AND tblArtikliProslost.Kolicina>0", acDialog, strSortiraj
    End If

Exit_cmdIzvestaj_Click:
    Exit Sub

Err_cmdIzvestaj_Click:
    MsgBox Err.Description
    Resume Exit_cmdIzvestaj_Click
    
End Sub

Private Sub PrikaziNaslov()

    Dim rstNaslov As New ADODB.Recordset
    If boolIzmena = False And lngRedniBroj = lngMaxRedniBroj Then
        rstNaslov.Open "SELECT SUM(Kolicina * ProdajnaCena) AS Vrednost FROM tblArtikli WHERE IDObjekat=" & _
        lngIDObjekat, cnnVeza, adOpenKeyset, adLockOptimistic
    Else
        rstNaslov.Open "SELECT SUM(Kolicina * ProdajnaCena) AS Vrednost FROM tblArtikliProslost WHERE IDObjekat=" & _
        lngIDObjekat, cnnVeza, adOpenKeyset, adLockOptimistic
    End If
    
    Form.Caption = "Stanje u objektu " & NazivObjekta & " na dan " & dateDatum & ": " & Format(rstNaslov!Vrednost, "currency")
    rstNaslov.Close

End Sub

Private Sub Izaberi()
    If lstPregled.ListIndex >= 0 Then
        lngIDArtikal = lstPregled.Column(0, lstPregled.ListIndex + 1)
        PronadjiArtikal
        If IsShiftKeyDown = False Then
            DoCmd.OpenForm "frmPromenaArtikla", , , , , acDialog
            If boolPonisti = False Then
                lstPregled.Requery
                PrikaziNaslov
            End If
        Else
            DoCmd.OpenForm "frmKartica", , , , , acDialog
        End If
    End If
End Sub

Private Sub lstPregled_KeyDown(KeyCode As Integer, Shift As Integer)
    If KeyCode = vbKeyReturn Then
        Izaberi
        lstPregled.SetFocus
    End If
