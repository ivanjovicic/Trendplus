Private Sub cmdDodaj_Click()
    
    If lstPronadji.ListIndex = -1 Then
        txtPronadji.SetFocus
        Exit Sub
    End If
    
    lngIDArtikal = lstPronadji.Column(0, lstPronadji.ListIndex + 1)
    If PronadjiArtikal = True Then
    End If
    
    For intBrojac = 1 To lstProdaja.ListCount - 1
        If lstProdaja.Column(0, intBrojac) = lngIDArtikal Then
            intKolicina = intKolicina - lstProdaja.Column(3, intBrojac)
        End If
    Next intBrojac
    
    intProdajaKolicina = CInt(txtKolicina.Value)
    
    If intKolicina < intProdajaKolicina Then
        MsgBox "Lager artikla " & strArtikal & " iznosi " & intKolicina & "!"
        Exit Sub
    End If
    
    lstProdaja.AddItem lngIDArtikal & ";" & intPLU & ";" & strArtikal & ";" & intProdajaKolicina & ";" & curProdajnaCena
    curIznos = curIznos + intProdajaKolicina * curProdajnaCena
    labIznos.Caption = "Iznos: " & curIznos
    
    txtPronadji.SetFocus
    
End Sub

    If lstProdaja.ListIndex >= 0 Then
        curIznos = curIznos - lstProdaja.Column(3, lstProdaja.ListIndex + 1) * lstProdaja.Column(4, lstProdaja.ListIndex + 1)
        labIznos.Caption = "Iznos: " & curIznos
        lstProdaja.RemoveItem (lstProdaja.ListIndex + 1)
    End If
End Sub

Private Sub cmdPotvrdi_Click()
    
    boolDesilaSePromena = True
    
    If (IsNull(txtDnevniIzvestaj.Value) Or txtDnevniIzvestaj.Value = "") And strTipPromene = "PRODAJA" Then
        MsgBox "Niste uneli broj dnevnog izvestaja!"
        Exit Sub
    Else
        If txtDnevniIzvestaj.Enabled = True Then
            strBrojKalkulacije = txtDnevniIzvestaj.Value
        End If
    End If
        
    If boolIzmena = True Then
        Dim rstPrivremena As New ADODB.Recordset
        If strTipPromene = "PRODAJA" Then
            PonistiProdaju lngRedniBroj
            rstPrivremena.Open "tblProdajaPrivremena", cnnVeza, adOpenKeyset, adLockOptimistic
        ElseIf strTipPromene = "POVRATNICA" Then
            PonistiPovracaj lngRedniBroj
            rstPrivremena.Open "tblPovratnicePrivremena", cnnVeza, adOpenKeyset, adLockOptimistic
        End If
        
        Do Until rstPrivremena.EOF
            boolNadjeno = False
            lngIDArtikal = rstPrivremena!IDArtikal
            For intBrojac = 1 To lstProdaja.ListCount - 1
                If lstProdaja.Column(0, intBrojac) = lngIDArtikal Then
                    boolNadjeno = True
                    Exit For
                End If
            Next intBrojac
            If boolNadjeno = False Then
                If PrikaziPromene_UPPP("PONISTI") = -1 Then
                    MsgBox "Nesto nije u redu...!"
                    Exit Sub
                End If
            End If
            rstPrivremena.MoveNext
        Loop
        rstPrivremena.Close
    End If
    
    lstPronadji.RowSource = ""
    
    If strTipPromene = "PRODAJA" Then
        ZapocniProdaju
    
        For intBrojac = 1 To lstProdaja.ListCount - 1
            lngIDArtikal = lstProdaja.Column(0, intBrojac)
            intProdajaKolicina = lstProdaja.Column(3, intBrojac)
            If ProdajaArtikla = False Then
                MsgBox "Doslo je do greske u unosu prodaje!"
                Exit Sub
            End If
        Next intBrojac
    
        PotvrdiProdaju
    
        MsgBox "Prodaja uspesno evidentirana!"
    
    ElseIf strTipPromene = "POVRATNICA" Then
        ZapocniPovracaj
    
        For intBrojac = 1 To lstProdaja.ListCount - 1
            lngIDArtikal = lstProdaja.Column(0, intBrojac)
            intPovracajKolicina = lstProdaja.Column(3, intBrojac)
            If PovracajArtikla = False Then
                MsgBox "Doslo je do greske u unosu povratnice!"
                Exit Sub
            End If
        Next intBrojac
    
        PotvrdiPovracaj
    
        MsgBox "Povratnica uspesno evidentirana!"
    End If
    
    DoCmd.Close
    
End Sub

Private Sub Form_Load()
    
    curIznos = 0

    lstProdaja.AddItem "IDArtikal;PLU;Artikal;Kolicina;Prodajna cena"
    
    If strTipPromene = "PRODAJA" Then
        Form.Caption = "Dnevni pazar u objektu " & NazivObjekta & " na dan " & dateDatum
    ElseIf strTipPromene = "POVRATNICA" Then
        Form.Caption = "Povracaj robe u " & strNazivDobavljaca & " iz objekta " & NazivObjekta & " na dan " & dateDatum
    End If
    
    If boolIzmena = True Then
        
        Dim rstPrivremena As New ADODB.Recordset
        
        If strTipPromene = "PRODAJA" Then
            txtDnevniIzvestaj.Value = strBrojKalkulacije
        
            DoCmd.RunSQL "DELETE * FROM tblProdajaPrivremena"
        
            Dim rstProdaja As New ADODB.Recordset
        
            rstProdaja.Open "SELECT * FROM tblProdaja WHERE IDDnevnik=" & lngIDDnevnik, cnnVeza, adOpenKeyset, adLockOptimistic
            rstPrivremena.Open "tblProdajaPrivremena", cnnVeza, adOpenKeyset, adLockOptimistic
                
            Do Until rstProdaja.EOF
                lngIDArtikal = rstProdaja!IDArtikal
                intProdajaKolicina = rstProdaja!Kolicina
            
                If PronadjiArtikal = True Then
                End If
            
                lstProdaja.AddItem CStr(lngIDArtikal) & ";" & CStr(intPLU) & ";" & strArtikal & ";" & intProdajaKolicina & ";" & curProdajnaCena
            
                rstPrivremena.AddNew
                rstPrivremena!IDArtikal = lngIDArtikal
                rstPrivremena!Kolicina = intProdajaKolicina
                rstPrivremena!ProdajnaCena = curProdajnaCena
                rstPrivremena.Update
                rstProdaja.MoveNext
                curIznos = curIznos + intProdajaKolicina * curProdajnaCena
            Loop
        
            rstPrivremena.Close
            rstProdaja.Close
                    
        End If
        
        If strTipPromene = "POVRATNICA" Then
        
            DoCmd.RunSQL "DELETE * FROM tblPovratnicePrivremena"
        
            Dim rstPovratnica As New ADODB.Recordset
        
            rstPovratnica.Open "SELECT * FROM tblPovratnice WHERE IDDnevnik=" & lngIDDnevnik, cnnVeza, adOpenKeyset, adLockOptimistic
            rstPrivremena.Open "tblPovratnicePrivremena", cnnVeza, adOpenKeyset, adLockOptimistic
                
            Do Until rstPovratnica.EOF
                lngIDArtikal = rstPovratnica!IDArtikal
                intPovracajKolicina = rstPovratnica!Kolicina
            
                If PronadjiArtikal = True Then
                End If

Private Sub txtPronadji_Change()
    
    If txtPronadji.Text = "" Or IsNull(txtPronadji.Text) Then Exit Sub
    intPrethodni = 0
    PretraziListu
    
End Sub

Private Sub PretraziListu()
    
    Dim strPronadji As String
    If Screen.ActiveControl Is txtPronadji Then
        strPronadji = UCase(txtPronadji.Text)
    Else
        strPronadji = UCase(txtPronadji.Value)
    End If
    
    For intBrojac = intPrethodni + 1 To lstPronadji.ListCount - 1
        If frmPronadji.Value = 1 Then
            If UCase(lstPronadji.Column(2, intBrojac)) = strPronadji Then
                lstPronadji.Selected(intBrojac) = True
                Exit For
            End If
            If UCase(lstPronadji.Column(2, intBrojac)) Like "*" & strPronadji & "*" Then
                lstPronadji.Selected(intBrojac) = True
                Exit For
            End If
        Else
            If UCase(lstPronadji.Column(1, intBrojac)) = strPronadji Then
                lstPronadji.Selected(intBrojac) = True
                Exit For
            End If
        End If
    Next intBrojac
    
    If intPrethodni = lstPronadji.ListIndex + 1 Then
        For intBrojac = 1 To lstPronadji.ListCount - 1
            If frmPronadji.Value = 1 Then
                If UCase(lstPronadji.Column(2, intBrojac)) = strPronadji Then
                    lstPronadji.Selected(intBrojac) = True
                    Exit For
                End If
                If UCase(lstPronadji.Column(2, intBrojac)) Like "*" & strPronadji & "*" Then
                    lstPronadji.Selected(intBrojac) = True
                    Exit For
                End If
            Else
                If UCase(lstPronadji.Column(1, intBrojac)) = strPronadji Then
                    lstPronadji.Selected(intBrojac) = True
                    Exit For
                End If
            End If
        Next intBrojac
    End If

    If lstPronadji.ListIndex > -1 Then
        lngIDArtikal = lstPronadji.Column(0, lstPronadji.ListIndex + 1)
        If PronadjiArtikal = True Then
        End If
        labStatusBar.Caption = " Artikal: " & strArtikal & ", PLU: " & intPLU & ", Lager: " & intKolicina & ", Naziv: " & NazivObuce & ", Cena: " & curProdajnaCena
    End If

End Sub
