Private Sub cmdDetaljaProvera_Click()
    Dim rstDnevnik As New ADODB.Recordset
    Dim prevDate As Date
    Dim pogresni As String
    
    Open "pogresni.txt" For Output As #67
    
    rstDnevnik.Open "SELECT * FROM tblDnevnikPromena WHERE IDObjekat = " & _
        CStr(lngIDObjekat) & " ORDER BY RedniBroj ASC", cnnVeza, adOpenKeyset, adLockOptimistic
    
    Do Until rstDnevnik.EOF
        
        If Not rstDnevnik.BOF Then
            If CLng(prevDate) > CLng(rstDnevnik!Datum) Then
                pogresni = CStr(rstDnevnik!RedniBroj)
                Print #67, pogresni
            End If
                
        End If
        prevDate = rstDnevnik!Datum
        rstDnevnik.MoveNext
    
    Loop
    
    Close #67
    
    rstDnevnik.Close
    
    'If MsgBox("Program ce pokusati oporavak osnovne tabele, moguc gubitak podataka! Nastavljate?", vbYesNo) = vbYes Then
    '    DetaljnaProvera
    'End If
End Sub

Private Sub dateGlavni_Exit(Cancel As Integer)
    Dim dateMaxDatum As Date
    On Error Resume Next
    dateMaxDatum = DMax("Datum", "tblDnevnikPromena")
    If dateGlavni.Text >= dateMaxDatum Then
        dateDatum = dateGlavni.Value
        If lngRedniBroj < lngMaxRedniBroj Then
            Sadasnjost lngRedniBroj
            lngRedniBroj = lngMaxRedniBroj
        End If
    Else
        If MsgBox("Datum koji ste izabrali je manji od datuma poslednjeg unosa u bazu!" & Chr(13) & _
                "Promena ce biti ubacena nakon poslednjeg unosa tog dana. Nastavljate?", vbOKCancel) = vbOK Then
            dateDatum = dateGlavni.Value
            On Error GoTo NajmanjiDatum
            lngRedniBroj = DMax("RedniBroj", "tblDnevnikPromena", "Datum <= DateValue('" & CStr(dateDatum) & "')") + 1
            VratiUnazad lngRedniBroj
            UbaciRedniBroj lngRedniBroj
        Else
            dateGlavni.Value = dateDatum
        End If
        
    End If
    lngIDSezona = Sezona(dateDatum)
    labGlavni.Caption = NazivSezone
    Exit Sub

NajmanjiDatum:
    lngRedniBroj = 1
    dateGlavni.Value = dateDatum
    Resume Next
End Sub

Private Sub Form_Load()
    DoCmd.SetWarnings False
    
    Set cnnVeza = CurrentProject.Connection
    boolDesilaSePromena = False
    cmbObjekat.Value = cmbObjekat.ItemData(0)

    UcitajPodesavanja

    lngIDObjekat = cmbObjekat.Value
    
    If Not IsNull(DMax("RedniBroj", "tblDnevnikPromena")) Then
        lngMaxRedniBroj = DMax("RedniBroj", "tblDnevnikPromena")
        dateDatum = DatumDnevnik
        If DateValue(dateDatum) < DateValue(Date) Then
            dateDatum = Date
        End If
    Else
        lngMaxRedniBroj = 0
        dateDatum = Date
    End If
    
    dateGlavni.Value = dateDatum
    lngRedniBroj = lngMaxRedniBroj
    boolIzmena = False
    lngIDSezona = Sezona(dateDatum)
    labGlavni.Caption = NazivSezone
    
End Sub
