Private Sub chkPrvaProdajna_Click()
    If chkPrvaProdajna.Value = False Then
        txtPrvaProdajna.Enabled = False
    Else
        txtPrvaProdajna.Enabled = True
    End If

Private Sub chkSnizeni_Click()
    If chkSnizeni.Value = False Then
        txtProcenat.Enabled = False
    Else
        txtProcenat.Enabled = True
    End If
End Sub



Private Sub cmdObradi_Click()
    Dim rstStatProdaja As New ADODB.Recordset
    Dim rstStatUneto As New ADODB.Recordset
    
    Dim strStatDob As String
    Dim strStatTip As String
    Dim strStatSezona As String
    Dim strDatumOd As String
    Dim strDatumDo As String
    Dim intUkupnoProdato As Integer
    Dim intUkupnoNabavljeno As Integer
    Dim curUkupnoProdato As Currency
    Dim curUkupnoNabavljeno As Currency
    
    strDatumOd = DatePart("m", DateValue(txtDatumOd.Value)) & "/" & DatePart("d", DateValue(txtDatumOd.Value)) & "/" & DatePart("yyyy", DateValue(txtDatumOd.Value))
    strDatumDo = DatePart("m", DateValue(txtDatumDo.Value)) & "/" & DatePart("d", DateValue(txtDatumDo.Value)) & "/" & DatePart("yyyy", DateValue(txtDatumDo.Value))
    
    If Not IsNull(cmbDobavljac.Value) Then
        strStatDob = " AND IDDobavljac=" & CStr(cmbDobavljac.Value)
    Else
        strStatDob = ""
    End If
    If Not IsNull(cmbTipObuce.Value) Then
        strStatTip = " AND IDTipObuce=" & CStr(cmbTipObuce.Value)
    Else
        strStatTip = ""
    End If
    If Not IsNull(cmbSezona.Value) Then
        strStatSezona = " AND IDSezona=" & CStr(cmbSezona.Value)
    Else
        strStatSezona = ""
    End If

    Dim strSQL As String
    
    strSQL = "SELECT * FROM tblDnevnikPromena WHERE IDObjekat=" & lngIDObjekat & " AND (DateValue(Datum)>DateValue(#" & strDatumOd & _
    "#) AND DateValue(Datum)<DateValue(#" & strDatumDo & "#))"
    CreateAccessQuery "qryDnevnik", strSQL
    
    strSQL = "SELECT * FROM tblArtikli WHERE IDObjekat=" & lngIDObjekat & strStatDob & strStatTip & strStatSezona
    CreateAccessQuery "qryArtikli", strSQL
    
    strSQL = "SELECT * FROM tblProdaja WHERE IDDnevnik IN (SELECT IDDnevnik FROM qryDnevnik)" & _
    " AND IDArtikal IN (SELECT IDArtikal FROM qryArtikli)"
    CreateAccessQuery "qryProdaja", strSQL
    
    rstStatProdaja.Open "SELECT IDArtikal, ProdajnaCena, SUM(Kolicina) AS Prodato FROM qryProdaja" & _
    " GROUP BY IDArtikal, ProdajnaCena ORDER BY IDArtikal", cnnVeza, adOpenKeyset, adLockOptimistic

    rstStatUneto.Open "SELECT IDArtikal, ProdajnaCena, Uneto FROM qryUnetoUkupno" & _
    " ORDER BY IDArtikal DESC", cnnVeza, adOpenDynamic, adLockOptimistic
    
    lstTopArtikli.RowSource = ""
    lstTopNabavljeno.RowSource = ""
    lstTopNabavljeno.AddItem "IDArtikal;Artikal;Dobavljac;Tip obuce;Nabavljeno;Nabavna cena"
    lstTopArtikli.AddItem "IDArtikal;Artikal;Dobavljac;Tip obuce;Prodato;Prodajna cena"
    
        Do Until rstStatProdaja.EOF
            lngIDArtikal = rstStatProdaja!IDArtikal
            If PronadjiArtikal = True And _
            (chkSnizeni.Value = False Or curNabavnaCena * Evro * (1 + (CSng(txtProcenat.Value)) / 100) <= rstStatProdaja!ProdajnaCena) And _
            (chkPrvaProdajna.Value = False Or curPrvaProdajnaCena * (1 - (CSng(txtPrvaProdajna.Value)) / 100) <= rstStatProdaja!ProdajnaCena) _
            Then
                strNazivDobavljaca = NazivDobavljaca
                strNazivObuce = NazivObuce
                If frmIzbor.Value = 2 Then
                    lstTopArtikli.AddItem lngIDArtikal & ";" & strArtikal & ";" & strNazivDobavljaca & ";" & strNazivObuce & _
                    ";" & rstStatProdaja!Prodato & ";" & CStr(rstStatProdaja!ProdajnaCena)
                ElseIf frmIzbor.Value = 1 Then
                    intUkupnoProdato = intUkupnoProdato + rstStatProdaja!Prodato
                ElseIf frmIzbor.Value = 3 Then
                    curUkupnoProdato = curUkupnoProdato + rstStatProdaja!Prodato * rstStatProdaja!ProdajnaCena * 0.847
                End If
            End If
            rstStatProdaja.MoveNext
        Loop
        
        If frmIzbor.Value = 1 Or frmIzbor.Value = 3 Then
            If IsNull(cmbDobavljac.Value) Then
                strNazivDobavljaca = "Svi dobavljaci"
            End If
            If IsNull(cmbTipObuce.Value) Then
                strNazivObuce = "Svi tipovi"
            End If
            If frmIzbor.Value = 1 Then
                lstTopArtikli.AddItem "1" & ";Ukupno;" & strNazivDobavljaca & ";" & strNazivObuce & ";" & intUkupnoProdato
            Else
                lstTopArtikli.AddItem "1" & ";Ukupno bez PDVa;" & strNazivDobavljaca & ";" & strNazivObuce & ";" & curUkupnoProdato
            End If
        End If
        

        
        Do Until rstStatUneto.EOF
            lngIDArtikal = rstStatUneto!IDArtikal
            If PronadjiArtikal = True And rstStatUneto!Uneto > 0 Then
                
                ' Ovo ispod nije tacno, moram da povezem prodaju i unos
                ' And (chkSnizeni.Value = False Or curNabavnaCena * Evro * (1 + (CSng(txtProcenat.Value)) / 100) <= rstStatUneto!ProdajnaCena) Then
                
                
                strNazivDobavljaca = NazivDobavljaca
                strNazivObuce = NazivObuce
                If frmIzbor.Value = 2 Then
                    lstTopNabavljeno.AddItem lngIDArtikal & ";" & strArtikal & ";" & strNazivDobavljaca & ";" & strNazivObuce & _
                    ";" & rstStatUneto!Uneto & ";" & CStr(curNabavnaCena * Evro)
                ElseIf frmIzbor.Value = 1 Then
                    intUkupnoNabavljeno = intUkupnoNabavljeno + rstStatUneto!Uneto
                ElseIf frmIzbor.Value = 3 Then
                    curUkupnoNabavljeno = curUkupnoNabavljeno + rstStatUneto!Uneto * curNabavnaCena * Evro
                End If
            End If
            rstStatUneto.MoveNext
        Loop
        
        If frmIzbor.Value = 1 Or frmIzbor.Value = 3 Then
            If IsNull(cmbDobavljac.Value) Then
                strNazivDobavljaca = "Svi dobavljaci"
            End If
            If IsNull(cmbTipObuce.Value) Then
                strNazivObuce = "Svi tipovi"
            End If
            If frmIzbor.Value = 1 Then
                lstTopNabavljeno.AddItem "1" & ";Ukupno;" & strNazivDobavljaca & ";" & strNazivObuce & ";" & intUkupnoNabavljeno
            Else
                lstTopNabavljeno.AddItem "1" & ";Ukupno;" & strNazivDobavljaca & ";" & strNazivObuce & ";" & curUkupnoNabavljeno
            End If
        End If
    
    rstStatProdaja.Close
    rstStatUneto.Close
    
    DoCmd.DeleteObject acQuery, "qryArtikli"
    DoCmd.DeleteObject acQuery, "qryDnevnik"
    DoCmd.DeleteObject acQuery, "qryProdaja"
    
End Sub
